Robot System Architecture

Pura end-to-end hardware communication flow: Central Processing Unit se lekar Sensors, Actuators aur Wireless telemetry tak.

1. High-Level AI & Compute
PRIMARY BRAIN

Raspberry Pi 4 Model B

Runs Linux OS & Python OpenCV pipelines for real-time Face Recognition, video telemetry, aur computer vision encodings.

VISION FEED

Pi Camera Module

Real-time optical feed capture for live identification, security scanning, aur remote video streaming over IP.

LOCAL LOGIC

Arduino Uno R3 / Mega

Low-level PWM generation, pin multiplexing, interrupt handling, and direct motor driver triggering.

2. Wireless & Sensor Bus
WI-FI TELEMETRY

NodeMCU ESP8266 V3

Hosts internal Web Server with direct IP access (192.168.x.x) for phone-controlled steering and automation toggles.

BLUETOOTH LINK

HC-05 RF Transceiver

Provides zero-latency wireless UART serial bridge for mobile app command execution and speech triggering.

AVOIDANCE

HC-SR04 Ultrasonic Sensor

Continuous ping-echo distance profiling (2cm - 400cm) for autonomous obstacle avoidance navigation.

3. Actuation & Output UI
MOTOR CONTROL

L298N Dual H-Bridge Driver

High-current drive for 4x BO 200 RPM gear motors enabling differential drive locomotion (Forward, Back, Spin).

FACIAL ANIMATRONICS

MG995 High-Torque Servos

Simulates physical facial expressions (eyebrow, mouth movements) corresponding to robot emotional states.

AUDIO & HUD

0.96" OLED & 10W Woofer

128x64 I2C animated eye graphics display paired with ISD1820 & PAM8403 amplified speech synthesizer.

// movementcontrol.ino - Project X Differential Drive Engine
#include <ESP8266WiFi.h>

const int ENA = D1;  // PWM Speed Control Left
const int IN1 = D2;  // Left Motor Forward
const int IN2 = D3;  // Left Motor Reverse
const int IN3 = D4;  // Right Motor Forward
const int IN4 = D5;  // Right Motor Reverse
const int ENB = D6;  // PWM Speed Control Right

void setup() {
  pinMode(ENA, OUTPUT);
  pinMode(ENB, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);
  Serial.begin(115200);
}

void moveForward(int speedVal) {
  analogWrite(ENA, speedVal);
  analogWrite(ENB, speedVal);
  digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
}

void stopMotors() {
  digitalWrite(IN1, LOW); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW); digitalWrite(IN4, LOW);
}

// babycradle.ino - IoT Motorized Automatic Cradle
#include <ESP8266WebServer.h>

ESP8266WebServer server(80);
const int MOTOR_PWM = D1;
bool isSwinging = false;

void handleRoot() {
  String html = "<h1>Baby Cradle Control</h1>";
  html += "<a href='/start'><button>Start Swing</button></a> ";
  html += "<a href='/stop'><button>Stop Swing</button></a>";
  server.send(200, "text/html", html);
}

void setup() {
  pinMode(MOTOR_PWM, OUTPUT);
  server.on("/", handleRoot);
  server.on("/start", []() { isSwinging = true; server.send(200, "text/plain", "Started"); });
  server.on("/stop", []() { isSwinging = false; server.send(200, "text/plain", "Stopped"); });
  server.begin();
}

void loop() {
  server.handleClient();
  analogWrite(MOTOR_PWM, isSwinging ? 180 : 0);
}
// armygate.ino - Automated Security Barrier System
#include <Servo.h>

Servo gateServo;
const int trigPin = 9;
const int echoPin = 10;
const int redLed = 6;
const int greenLed = 7;

void setup() {
  gateServo.attach(11);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  gateServo.write(0); // Gate Closed Default
}

void loop() {
  long duration, distance;
  digitalWrite(trigPin, LOW); delayMicroseconds(2);
  digitalWrite(trigPin, HIGH); delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  duration = pulseIn(echoPin, HIGH);
  distance = (duration / 2) / 29.1;

  if (distance < 20) { // Authorized Vehicle Detected
    gateServo.write(90); // Raise Barrier
    delay(5000);
    gateServo.write(0);  // Lower Barrier
  }
}
# face_recognize.py - Raspberry Pi 4 Face Vision Classifier
import cv2
import pickle

print("[INFO] loading encodings + face detector...")
data = pickle.loads(open("encodings.pickle", "rb").read())
detector = cv2.CascadeClassifier("haarcascade_frontalface_default.xml")

cap = cv2.VideoCapture(0)
while True:
    ret, frame = cap.read()
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    faces = detector.detectMultiScale(gray, scaleFactor=1.1, minNeighbors=5)
    for (x, y, w, h) in faces:
        cv2.rectangle(frame, (x, y), (x+w, y+h), (0, 255, 255), 2)
        cv2.putText(frame, "AUTHORIZED USER", (x, y-10), 
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 255), 2)
    cv2.imshow("Project X - Vision Stream", frame)
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break
Specifications & Modules

Hardware Inventory

Sabhi sensors, drivers, compute boards, aur electronic parts jo robot aur projects me integrate kiye gaye hain.

Compute Board

Raspberry Pi 4 (4GB)

Master AI brain for high-speed image processing, OpenCV facial recognition model, and web daemon hosting.

Microcontroller

Arduino Uno R3 / Mega

Primary logic controller managing servo kinematics, sensor readouts, and motor actuation timings.

Wireless IoT

NodeMCU ESP8266 V3

Integrated Wi-Fi module enabling custom mobile web server control and local network telemetry.

Communication

HC-05 Bluetooth Module

Serial RF bridge providing seamless Android terminal connectivity for speech synthesis and steering.

Motor Driver

L298N Dual H-Bridge

Dual DC motor driver with heat-sink capable of up to 2A per channel for high-torque wheel locomotion.

Animatronics

MG995 Metal Servos

High-torque metal gear servo motors providing precise mechanical movement for robot facial expressions.

Visual Display

0.96" I2C OLED (128x64)

High-contrast monochrome display showing eyes animation, battery status, and IP scanning telemetry.

Audio Synthesis

PAM8403 Amp + 10W Woofer

5V stereo amplifier module driving a 3-inch 4Ω woofer speaker for loud, clear Text-to-Speech audio feedback.

Proximity Sensor

HC-SR04 Ultrasonic

Non-contact sonar ranging module offering obstacle detection from 2cm up to 4 meters.

Power Delivery

18650 Li-Ion Cell Pack

High-discharge rechargeable 3.7V / 12V lithium-ion battery array powering independent logic and motor buses.

Propulsion

BO 200 RPM Gear Motors

Durable dual-shaft DC motors coupled with rubber grip wheels for stable forward/reverse navigation.

Sound & Record

ISD1820 Voice Board

Audio playback and voice recording module for pre-stored safety alerts and verbal robotic responses.
